    Integration

    Drop js/avo.js + p5.js into VoceSpace. Each client renders Avos locally from 5 serializable params — no images, no video tracks.

    2 · Store params per user

    Add avoParams to the user profile and broadcast it in room state.

    {
  name: 'Han',      // display name; seeds personality
  variant: 0,       // re-roll look without renaming
  hue: 193,         // from Avo.PALETTE only (theme-harmonic)
  style: 'blob',    // 'blob' | 'ring' | 'wave'
  energy: 0.6       // 0.1–1, voice reactivity strength
}

    3 · Mount in the participant tile

    Replace the gray silhouette with an empty holder div sized to the tile.

    <div class="participant-tile">
  <div class="avo-holder"></div>

</div>
    const avo = Avo.mount(tile.querySelector('.avo-holder'), {
  params: participant.avoParams,
  scale: 0.62,
  getLevel: () => participant.audioLevel,  // 0..1 per participant (LiveKit etc.)
  onSpeaking: (s) => tile.classList.toggle('speaking', s),
  interactive: participant.isLocal,        // hover / pet / click for local user only
});

// lifecycle
avo.setParams(newParams);  // room state updated

    4 · Settings page

    This page is the template. On every control change, call avo.setParams(params) and save to your API (replace demo Avo.saveParams() / localStorage). 

    async function apply(patch) {
  params = { ...params, ...patch };
  avo.setParams(params);           // live preview
}

async function saveAvo() {
  await api.patchUser({ avoParams: params });  // active avatar for room
  // optional: Avo.saveToLibrary(params) for preset slots in demo
}

// shuffle: Avo.randomParams(name)  ·  colors: Avo.PALETTE  ·  styles: Avo.STYLES
// backup: Avo.exportToFile(params)  ·  Avo.importFromFile(file)

    5 · Save & load

    The demo persists to localStorage; in VoceSpace, swap those calls for your API — the data shapes stay the same. Always run received or imported params through Avo.normalizeParams(). 

    // active avatar — what the room renders for this user
Avo.saveParams(params);                       // demo · prod: api.patchUser({ avoParams })
const params = Avo.loadParams() ?? Avo.DEFAULTS;  // on settings/room load

// saved library (the preset slots on this page, max 12)
const entry = Avo.saveToLibrary(params, label);   // saves + sets active
Avo.listSaved();                                  // [{ id, label, params, savedAt }]
Avo.deleteSaved(entry.id);
// "Load" = Avo.normalizeParams(entry.params) → avo.setParams() + save active

// static preview for lists (one frame, no animation loop)
const png = await Avo.snapshot(params, 168);      // PNG dataURL

// share / backup as .avo.json
Avo.exportToFile(params);
const imported = await Avo.importFromFile(file);  // validated + clamped
